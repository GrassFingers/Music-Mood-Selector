# ============================================================
#  SAFE AUDIO FEATURES FETCHER (FIX FOR 403 ERRORS)
# ============================================================
def get_audio_features_safe(sp, track_ids, batch_size=25):
    """
    Spotify 403s if the ?ids= URL becomes too long.
    This fetches audio features in small safe batches.
    """
    all_features = []
    print(f"\n🎛 Fetching audio features in safe batches ≤ {batch_size}...")

    for i in range(0, len(track_ids), batch_size):
        batch = track_ids[i:i + batch_size]
        print(f"  • Batch {i//batch_size + 1} → {len(batch)} tracks")

        try:
            features = sp.audio_features(batch)
            if features:
                all_features.extend(features)
        except Exception as e:
            print(f"❌ Error fetching batch {i}-{i+len(batch)-1}: {e}")
            continue

    print(f"✔ Total audio features returned: {len(all_features)}")
    return [f for f in all_features if f]


# ============================================================
#  FETCH ALL TRACKS FROM A PLAYLIST (PAGINATED)
# ============================================================
def get_playlist_track_ids(sp, playlist_url):
    print("\n📥 Fetching playlist (paginated)...")
    playlist_id = playlist_url.split("/")[-1].split("?")[0]

    track_ids = []
    limit = 100
    offset = 0

    while True:
        data = sp.playlist_items(playlist_id, limit=limit, offset=offset)
        items = data["items"]
        print(f"  • Fetched {len(items)} items at offset {offset}")

        for item in items:
            track = item.get("track")
            if track and track.get("id"):
                track_ids.append(track["id"])

        if len(items) < limit:
            break

        offset += limit

    print(f"✔ Total Spotify track IDs collected: {len(track_ids)}")
    return track_ids

def filter_tracks_by_mood(features, mood):
    """
    Very simple example logic.
    You can customize thresholds.
    """
    print(f"\n🎚 Applying mood filter: {mood}")

    if mood == "Happy":
        return [f for f in features if f["valence"] > 0.6 and f["energy"] > 0.6]

    elif mood == "Sad":
        return [f for f in features if f["valence"] < 0.4]

    elif mood == "Chill":
        return [f for f in features if f["energy"] < 0.4]

    elif mood == "Energetic":
        return [f for f in features if f["energy"] > 0.7]

    return features

# =====================================================
# SPOTIFY CLIENT
# =====================================================
def get_spotify_client():
    sp = spotipy.Spotify(auth_manager=SpotifyOAuth(
        client_id=CLIENT_ID,
        client_secret=CLIENT_SECRET,
        redirect_uri=REDIRECT_URI,
        scope=SCOPE
    ))
    return sp


# =====================================================
# FIND MBID FROM MUSICBRAINZ
# =====================================================
def find_mbid(track_name, artist_name):
    try:
        result = musicbrainzngs.search_recordings(
            recording=track_name,
            artist=artist_name,
            limit=1
        )
        if result['recording-list']:
            return result['recording-list'][0]['id']
    except:
        pass
    return None


# =====================================================
# CALL ACOUSTICBRAINZ
# =====================================================
def get_acousticbrainz_mood(mbid):
    url = f"https://acousticbrainz.org/{mbid}/high-level"
    try:
        r = requests.get(url, timeout=3)
        if r.status_code == 200:
            return r.json()
    except:
        return None
    return None


# =====================================================
# MOOD MATCHER
# =====================================================
def track_matches_mood(acb_data, desired_mood):
    if not acb_data or "highlevel" not in acb_data:
        return False

    moods = acb_data["highlevel"]

    mood_map = {
        "happy": moods["mood_happy"]["probability"],
        "sad": moods["mood_sad"]["probability"],
        "relaxed": moods["mood_relaxed"]["probability"],
        "aggressive": moods["mood_aggressive"]["probability"],
        "party": moods["mood_party"]["probability"],
        "acoustic": moods["mood_acoustic"]["probability"]
    }

    if desired_mood not in mood_map:
        return False

    # Threshold
    return mood_map[desired_mood] > 0.5


# =====================================================
# FILTER PLAYLIST
# =====================================================
def filter_playlist_by_mood(sp, playlist_id, desired_mood, log_callback):
    tracks = sp.playlist_tracks(playlist_id)["items"]
    matched_tracks = []

    for idx, item in enumerate(tracks):
        track = item["track"]
        name = track["name"]
        artist = track["artists"][0]["name"]
        spotify_id = track["id"]

        log_callback(f"Analyzing {idx+1}/{len(tracks)}: {name} — {artist}")

        # 1. MBID
        mbid = find_mbid(name, artist)
        if not mbid:
            log_callback("  ❌ No MBID found")
            continue

        # 2. AcousticBrainz
        acb = get_acousticbrainz_mood(mbid)
        if not acb:
            log_callback("  ❌ No AcousticBrainz data")
            continue

        # 3. Mood match?
        if track_matches_mood(acb, desired_mood):
            log_callback("  ✅ MATCHES mood!")
            matched_tracks.append(spotify_id)
        else:
            log_callback("  ❌ Does NOT match mood")

        time.sleep(1)

    return matched_tracks

def create_filtered_playlist(sp, user_id, original_name, mood, track_ids):
    new_name = f"{original_name} — {mood} mood"
    playlist = sp.user_playlist_create(user_id, new_name)

    for i in range(0, len(track_ids), 100):
        sp.playlist_add_items(playlist["id"], track_ids[i:i+100])

    return playlist["external_urls"]["spotify"]

def run_mood_filter():
    playlist_url = playlist_entry.get().strip()
    mood = mood_combo.get().strip()

    if playlist_url == "" or mood == "":
        messagebox.showerror("Error", "Please enter a playlist URL and select a mood.")
        return

    log_text.delete(1.0, tk.END)
    log_text.insert(tk.END, "Logging in to Spotify...\n")

    def log(msg):
        log_text.insert(tk.END, msg + "\n")
        log_text.see(tk.END)

    def process():
        try:
            sp = get_spotify_client()
            playlist_id = playlist_url.split("/")[-1].split("?")[0]
            playlist = sp.playlist(playlist_id)
            original_name = playlist["name"]
            user_id = sp.current_user()["id"]

            log("Filtering playlist...")

            matches = filter_playlist_by_mood(sp, playlist_id, mood, log)

            log(f"\nFound {len(matches)} matching tracks.")

            if matches:
                url = create_filtered_playlist(sp, user_id, original_name, mood, matches)
                log(f"\n🎉 Playlist created:\n{url}")
            else:
                log("\nNo tracks match the selected mood.")
        except Exception as e:
            log(f"\nError: {str(e)}")

    threading.Thread(target=process).start()
